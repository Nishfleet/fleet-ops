# --- Root-only files under /etc -----------------------------------------
#
# The live unit runs as root, so it reads all of /etc. These are excluded on
# purpose, not because they are unreadable:

# Local root password hashes and equivalent. A leaked backup is an offline
# cracking target and a shadow file cannot be restored usefully anyway.
/etc/shadow
/etc/shadow-
/etc/gshadow
/etc/gshadow-
/etc/opasswd

# Host private keys: a stolen copy lets an attacker impersonate the box.
# Regenerate them after a restore; known_hosts entries change once.
/etc/ssh/ssh_host_*_key

# sudo policy and the account database backups it is compared against.
/etc/sudoers.d/
/etc/passwd-
/etc/group-
/etc/subuid-
/etc/subgid-

# TLS private keys.
/etc/ssl/private/

# The restic repository password and the R2 keys themselves. They are stored
# inside the directory this list protects, so backing them up would put the
# decryptor next to the ciphertext. They are root-only (mode 0700) and are
# held outside the backup on purpose: a restore asks the operator for them,
# the same way it asks for RESTIC_PASSWORD.
/etc/restic/
